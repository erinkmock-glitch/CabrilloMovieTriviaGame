    // ========= WEB AUDIO MUSIC ENGINE =========
    let audioCtx = null;
    let activeAudioNodes = [];

    function initAudio() {
      if (!audioCtx) {
        const AC = window.AudioContext || window.webkitAudioContext;
        audioCtx = new AC();
      }
    }

    function registerNode(node) {
      if (node) activeAudioNodes.push(node);
    }

    function stopAllMusic() {
      activeAudioNodes.forEach(node => {
        try {
          if (node.stop) node.stop();
          node.disconnect();
        } catch (e) {}
      });
      activeAudioNodes = [];
    }

    // Long, soft pad
    function startAmbientPad(baseFreq, volume = 0.4) {
      initAudio();
      const now = audioCtx.currentTime;

      for (let i = 0; i < 3; i++) {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = "sine";
        osc.frequency.value = baseFreq * (1 + i * 0.05);

        gain.gain.setValueAtTime(0, now);
        gain.gain.linearRampToValueAtTime(volume / (i + 1), now + 2);
        gain.gain.linearRampToValueAtTime(volume / (i + 1), now + 18);
        gain.gain.linearRampToValueAtTime(0, now + 22);

        osc.connect(gain).connect(audioCtx.destination);
        osc.start(now);
        osc.stop(now + 22);

        registerNode(osc);
        registerNode(gain);
      }
    }

    // Bell-like tinkles
    function startBellPad(baseFreq, volume = 0.25) {
      initAudio();
      const now = audioCtx.currentTime;

      for (let i = 0; i < 4; i++) {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = "triangle";
        osc.frequency.value = baseFreq * (1 + i * 0.07);

        const start = now + i * 1.5;
        gain.gain.setValueAtTime(0, start);
        gain.gain.linearRampToValueAtTime(volume, start + 0.3);
        gain.gain.linearRampToValueAtTime(0.01, start + 2);

        osc.connect(gain).connect(audioCtx.destination);
        osc.start(start);
        osc.stop(start + 3);

        registerNode(osc);
        registerNode(gain);
      }
    }

    // Simple synthwave-ish bass + chord
    function startSynthwave() {
      initAudio();
      const now = audioCtx.currentTime;

      const bass = audioCtx.createOscillator();
      const bassGain = audioCtx.createGain();
      bass.type = "sawtooth";
      bass.frequency.value = 110;

      bassGain.gain.setValueAtTime(0, now);
      bassGain.gain.linearRampToValueAtTime(0.18, now + 1);
      bassGain.gain.linearRampToValueAtTime(0.18, now + 18);
      bassGain.gain.linearRampToValueAtTime(0, now + 22);

      bass.connect(bassGain).connect(audioCtx.destination);
      bass.start(now);
      bass.stop(now + 22);

      registerNode(bass);
      registerNode(bassGain);

      const chordFreqs = [220, 277, 330]; // A minor-ish
      chordFreqs.forEach((f, i) => {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = "square";
        osc.frequency.value = f;

        gain.gain.setValueAtTime(0, now + i * 0.2);
        gain.gain.linearRampToValueAtTime(0.12, now + 2 + i * 0.2);
        gain.gain.linearRampToValueAtTime(0.12, now + 18);
        gain.gain.linearRampToValueAtTime(0, now + 22);

        osc.connect(gain).connect(audioCtx.destination);
        osc.start(now);
        osc.stop(now + 22);

        registerNode(osc);
        registerNode(gain);
      });
    }

    // Organ-like pad

    // Simple vintage-y melody
    function startVintageMelody() {
      initAudio();
      const now = audioCtx.currentTime;

      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = "sine";

      const notes = [330, 392, 440, 392, 330, 294, 262, 294];

        const t = now + i * 1.2;
        osc.frequency.setValueAtTime(freq, t);
        gain.gain.setValueAtTime(0, t);
        gain.gain.linearRampToValueAtTime(0.2, t + 0.2);
        gain.gain.linearRampToValueAtTime(0.03, t + 1);
      });

      osc.connect(gain).connect(audioCtx.destination);
      osc.start(now);
      osc.stop(now + notes.length * 1.2 + 2);

      registerNode(osc);
      registerNode(gain);
    }

    function playDecadeMusic(decade) {
      stopAllMusic();

      // Each call creates a long ambient "pass" (~20s).
      // If the player stays longer, it's still fine; we just restart on decade changes.
      switch (decade) {
        case "2000s":
          startAmbientPad(220, 0.35);
          break;
        case "1990s":
          startBellPad(440, 0.22);

          break;

          startWarmHum(110);
          break;
        case "1960s":
          startOrganPad(330);
          break;
        case "1950s":
          startVintageMelody();
          break;
      }
    }

    // ========= HELPERS =========
    function normalize(str) {
      return str
        .toLowerCase()
        .trim()
        .replace(/[^a-z0-9 ]/g, "")
        .replace(/^the\s+/, ""); // ignore leading "the"
    }

    function isCorrectAnswer(userInput, acceptableAnswers) {
      const normUser = normalize(userInput);
      return acceptableAnswers.some(ans => normalize(ans) === normUser);
    }

    // ========= GAME FLOW =========
    function startGame() {
      currentRoundIndex = 0;
      currentQuestionIndex = 0;
      correctThisRound = 0;
      awaitingAnswer = true;
      gameStarted = true;

      restartBtn.classList.add("hidden");
      submitBtn.disabled = false;
      nextBtn.disabled = true;
      answerRow.classList.remove("hidden");

      feedbackText.textContent = "";
      feedbackText.className = "feedback";
      roundSummaryText.textContent = "";

      updateRoundBadges();
      showQuestion();
      playDecadeMusic(rounds[currentRoundIndex].decade);
    }

    function updateRoundBadges() {
      const totalRounds = rounds.length;
      const roundNumber = currentRoundIndex + 1;
      const decade = rounds[currentRoundIndex].decade;
      roundBadge.textContent = `Round ${roundNumber} / ${totalRounds}`;
      decadeBadge.textContent = decade;
    }

    function showQuestion() {
      const round = rounds[currentRoundIndex];
      const question = round.questions[currentQuestionIndex];
      const qNumber = currentQuestionIndex + 1;
      const totalQ = round.questions.length;

      progressText.textContent = `Decade: ${round.decade} — Question ${qNumber} of ${totalQ} — Need at least 3 correct to pass.`;
      quoteElement.textContent = question.quote;

      answerInput.value = "";
      answerInput.disabled = false;
      answerInput.focus();

      feedbackText.textContent = "";
      feedbackText.className = "feedback";
      roundSummaryText.textContent = "";

      awaitingAnswer = true;
      submitBtn.disabled = false;
      nextBtn.disabled = true;
      nextBtn.textContent = "Next ➡️";
    }

    function checkAnswer() {
      const userAnswer = answerInput.value;
      if (!userAnswer.trim()) {
        feedbackText.textContent = "Type a movie title before submitting.";
        feedbackText.className = "feedback incorrect-text";
        return;
      }

      const round = rounds[currentRoundIndex];
      const question = round.questions[currentQuestionIndex];

      const correct = isCorrectAnswer(userAnswer, question.answers);
      if (correct) {
        correctThisRound++;
        feedbackText.textContent = "Correct! 🎉";
        feedbackText.className = "feedback correct-text";
      } else {
        const mainAnswer = question.answers[0];
        feedbackText.textContent = `Not quite! The movie was "${mainAnswer}".`;
        feedbackText.className = "feedback incorrect-text";
      }

      answerInput.disabled = true;
      submitBtn.disabled = true;
      nextBtn.disabled = false;
      awaitingAnswer = false;
    }

    function handleNext() {
      const round = rounds[currentRoundIndex];
      const totalQ = round.questions.length;

      if (currentQuestionIndex < totalQ - 1) {
        currentQuestionIndex++;
        showQuestion();
      } else {
        endRound();
      }
    }

    function endRound() {
      const needed = 3;
      const totalQ = rounds[currentRoundIndex].questions.length;
      const decade = rounds[currentRoundIndex].decade;
      const passed = correctThisRound >= needed;

      answerInput.disabled = true;
      submitBtn.disabled = true;
      nextBtn.disabled = false;
      awaitingAnswer = false;

      roundSummaryText.textContent = `You got ${correctThisRound} out of ${totalQ} correct in the ${decade}.`;

      if (!passed) {
        feedbackText.textContent = `You needed at least ${needed} to pass the ${decade}. Game over. 😢`;
        feedbackText.className = "feedback incorrect-text";
        nextBtn.classList.add("hidden");
        restartBtn.classList.remove("hidden");
        stopAllMusic();
      } else {
        if (currentRoundIndex === rounds.length - 1) {
          feedbackText.textContent = `You passed the ${decade} and completed all decades. You win! 🏆`;
          feedbackText.className = "feedback correct-text";
          nextBtn.classList.add("hidden");
          restartBtn.classList.remove("hidden");
          stopAllMusic();
        } else {
          const nextDecade = rounds[currentRoundIndex + 1].decade;
          feedbackText.textContent = `You passed the ${decade}! Next up: the ${nextDecade}.`;
          feedbackText.className = "feedback correct-text";

          nextBtn.textContent = "Start next decade ⏭️";
          nextBtn.onclick = () => {
            currentRoundIndex++;
            currentQuestionIndex = 0;
            correctThisRound = 0;
            updateRoundBadges();
            nextBtn.onclick = handleNext; // restore default
            showQuestion();
            playDecadeMusic(rounds[currentRoundIndex].decade);
          };
        }
      }
    }

    // ========= EVENT LISTENERS =========
    submitBtn.addEventListener("click", () => {
      if (awaitingAnswer && gameStarted) {
        checkAnswer();
      }
    });

    nextBtn.addEventListener("click", () => {
      if (!awaitingAnswer && gameStarted) {
        handleNext();
      }
    });

    restartBtn.addEventListener("click", () => {
      nextBtn.classList.remove("hidden");
      restartBtn.classList.add("hidden");
      nextBtn.onclick = handleNext;
      startGame();
    });

    answerInput.addEventListener("keydown", (e) => {
      if (e.key === "Enter") {
        e.preventDefault();
        if (!gameStarted) return;
        if (awaitingAnswer && !submitBtn.disabled) {
          checkAnswer();
        } else if (!awaitingAnswer && !nextBtn.disabled) {
          handleNext();
        }
      }
    });

    introStartBtn.addEventListener("click", () => {
      introOverlay.classList.add("hidden");
      startGame();
    });
  </script>
